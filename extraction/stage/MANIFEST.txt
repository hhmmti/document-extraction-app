# values.py
#   `from schemas import (` -> `from .schemas import (`

#   `path=path`, `slug=path.stem.replace("-pass-a", "")` -> `path=None`, `slug=""`
#     No file. `slug` was read in one place, the fallback `source_document or slug`

#   property `family_code` (file-name prefix) -> field `family_code`, set from the caller's family
#     The family now comes from Detect, checked against frontmatter vendor_family.
#   `read_all(reports_dir)` removed: it read a directory; callers pass texts.
